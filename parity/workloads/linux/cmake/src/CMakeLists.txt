cmake_minimum_required(VERSION 3.20)
project(aap_parity VERSION 1.0.0 LANGUAGES C CXX)
add_library(parity STATIC parity.c)
add_executable(app main.cpp)
target_link_libraries(app PRIVATE parity)
target_compile_features(app PRIVATE cxx_std_17)
enable_testing()
add_test(NAME app_runs COMMAND app)
set_tests_properties(app_runs PROPERTIES PASS_REGULAR_EXPRESSION "parity_add.2, 3. = 5")
install(TARGETS app RUNTIME DESTINATION bin)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_FILE_NAME aap-parity)
include(CPack)
